CREATE DATABASE SlahTune
GO
USE SlahTune

CREATE TABLE Usuario
(
	ID INT IDENTITY PRIMARY KEY NOT NULL,
	Nome VARCHAR(30) NOT NULL,
	Senha VARCHAR(200)NOT NULL,
	Email VARCHAR(100)NOT NULL,
	Foto IMAGE NOT NULL
)

CREATE TABLE Nota
(
	ID INT IDENTITY PRIMARY KEY NOT NULL,
	Nota VARCHAR(25) NOT NULL,
	Frequencia VARCHAR(MAX) NOT NULL
)

CREATE TABLE Artista
(
	ID INT IDENTITY PRIMARY KEY NOT NULL,
	Nome VARCHAR(75) NOT NULL
)

CREATE TABLE Musica
(
	ID INT IDENTITY PRIMARY KEY NOT NULL,
	Nome VARCHAR(150) NOT NULL
)

CREATE TABLE Musica_Has_Artista
(
	IDMusica INT REFERENCES Musica (ID) NOT NULL,
	IDArtista INT REFERENCES Artista (ID) NOT NULL
)

CREATE TABLE Tablatura
(
	ID INT IDENTITY PRIMARY KEY NOT NULL,
	Identificador VARCHAR(MAX),
	IDUsuario INT REFERENCES Usuario(ID),
	IDMusica INT REFERENCES Musica (ID) NOT NULL,
)
